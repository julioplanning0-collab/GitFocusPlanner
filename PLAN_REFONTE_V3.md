## 🎯 OBJECTIFS DE LA REFONTE

### Pourquoi cette refonte ?

**Problème actuel**: Le code actuel utilise un système hybride qui mélange:
- Manipulation de chaînes de date/heure (complexe, erreurs de parsing)
- Calculs directs sur créneaux libres (perd la continuité temporelle)
- Insertion manuelle d'obstacles (code répétitif, difficile à maintenir)

**Solution**: Système de timeline linéaire avec temps relatif
- Arithmétique simple (additions de minutes entières)
- Placement séquentiel avec détection de collisions
- Architecture 2-step claire (ORDERING puis TIME CALCULATION)
- Drag & drop natif (déplacement sur axe linéaire)

### Garanties de sécurité

**Commits réguliers**: Après CHAQUE phase complète
**Tests à chaque étape**: Validation avant passage à la phase suivante
**Rollback clair**: `git revert` si problème, pas de perte de données
**Backup specs**: Ce document + TIMELINE_LINEAR_SPEC_BACKUP.md + flux_planning_BACKUP.md

```javascript
/**
 * Convert Date object to minutes from planningStartTime.
 * Supports multi-day (negative values for past, large values for future).
 *
 * @param {Date} dateTime - Absolute time to convert
 * @param {Date} planningStart - Reference time (minute 0)
 * @returns {number} Minutes from planningStart (can be negative)
 *
 * @example
 * const start = new Date('2025-11-05T14:00:00');
 * const time = new Date('2025-11-05T16:30:00');
 * dateTimeToMinutes(time, start); // 150 (2h30)
 */
function dateTimeToMinutes(dateTime, planningStart) {
    return Math.floor((dateTime - planningStart) / 60000);
}

/**
 * Convert minutes from planningStartTime to {date, time} object.
 *
 * @param {number} minutes - Minutes from planningStart
 * @param {Date} planningStart - Reference time (minute 0)
 * @returns {{date: string, time: string}} ISO date + HH:MM time
 *
 * @example
 * const start = new Date('2025-11-05T14:00:00');
 * minutesToDateTime(150, start); // {date: '2025-11-05', time: '16:30'}
 */
function minutesToDateTime(minutes, planningStart) {
    const absoluteTime = new Date(planningStart.getTime() + minutes * 60000);
    const date = absoluteTime.toISOString().split('T')[0]; // "2025-11-05"
    const time = absoluteTime.toTimeString().substring(0, 5); // "16:30"
    return {date, time};
}

/**
 * Convert temps_morts CSV data to linear obstacles.
 * Filters out past obstacles (endMinute < 0) and very far future (> 30 days).
 *
 * @param {Array<Object>} tempsMorts - CSV data [{date, heure_debut, heure_fin, titre}]
 * @param {Date} planningStart - Reference time
 * @returns {Array<Obstacle>} Linear obstacles sorted by startMinute
 */
function convertTempsMortsToLinear(tempsMorts, planningStart) {
    const obstacles = [];
    const MAX_FUTURE_MINUTES = 43200; // 30 days

    for (const tm of tempsMorts) {
        const startDateTime = new Date(`${tm.date}T${tm.heure_debut}:00`);
        const endDateTime = new Date(`${tm.date}T${tm.heure_fin}:00`);

        const startMinute = dateTimeToMinutes(startDateTime, planningStart);
        const endMinute = dateTimeToMinutes(endDateTime, planningStart);

        // Skip past obstacles
        if (endMinute < 0) continue;

        // Skip very far future obstacles
        if (startMinute > MAX_FUTURE_MINUTES) continue;

        obstacles.push({
            startMinute,
            endMinute,
            type: 'temps_mort',
            originalData: tm
        });
    }

    // Sort by startMinute ascending
    return obstacles.sort((a, b) => a.startMinute - b.startMinute);
}
```

```javascript
/**
 * Calculate planning start time: now + 15min, rounded up to next 5min.
 * Updates state.planningStartTime.
 */
function initializePlanningStart() {
    const now = new Date();
    const future = new Date(now.getTime() + 15 * 60000); // +15 minutes

    // Round up to next 5-minute mark
    const minutes = future.getMinutes();
    const roundedMinutes = Math.ceil(minutes / 5) * 5;

    // Handle overflow (58 → 60 → next hour)
    if (roundedMinutes >= 60) {
        future.setHours(future.getHours() + 1);
        future.setMinutes(0);
    } else {
        future.setMinutes(roundedMinutes);
    }

    future.setSeconds(0);
    future.setMilliseconds(0);

    state.planningStartTime = future;

    // Update UI input
    const timeInput = document.getElementById('planning-start-time');
    if (timeInput) {
        timeInput.value = future.toTimeString().substring(0, 5); // "HH:MM"
    }

    console.log(`Planning start time: ${future.toLocaleString()}`);
    return future;
}

/**
 * Load temps_morts for selected date, convert to linear obstacles.
 * Updates state.timeline.obstacles.
 */
async function initializeTimelineObstacles() {
    const selectedDate = document.getElementById('planning-date').value; // "YYYY-MM-DD"

    try {
        // Load temps_morts from API
        const response = await fetch(`/api/v2/gitfocus/temps-morts?date=${selectedDate}`);
        const data = await response.json();

        if (!data.success) {
            console.error('Failed to load temps_morts:', data.error);
            state.timeline.obstacles = [];
            return;
        }

        // Convert to linear obstacles
        state.timeline.obstacles = convertTempsMortsToLinear(
            data.temps_morts || [],
            state.planningStartTime
        );

        console.log(`Loaded ${state.timeline.obstacles.length} obstacles`);

    } catch (error) {
        console.error('Error loading temps_morts:', error);
        state.timeline.obstacles = [];
    }
}
```

### PHASE 3: Core Placement Algorithm - findNextFreeSlot (4 heures)

**Objectif**: Fonction centrale de placement avec détection de collisions

**Fonction à créer**:

```javascript
/**
 * Find next free slot on timeline that doesn't overlap obstacles or existing tasks.
 *
 * @param {number} startMinute - Desired start position (minutes from planningStart)
 * @param {number} durationMinutes - Task duration
 * @param {Object} timeline - {tasks: Task[], obstacles: Obstacle[]}
 * @returns {number|null} Position of next free slot, or null if none found in 30 days
 */
function findNextFreeSlot(startMinute, durationMinutes, timeline) {
    const MAX_SEARCH = 43200; // 30 days in minutes
    let candidateStart = startMinute;

    while (candidateStart < MAX_SEARCH) {
        const candidateEnd = candidateStart + durationMinutes;
        let hasOverlap = false;
        let jumpTo = candidateStart;

        // Check obstacles
        for (const obstacle of timeline.obstacles) {
            if (overlaps(candidateStart, candidateEnd, obstacle.startMinute, obstacle.endMinute)) {
                jumpTo = Math.max(jumpTo, obstacle.endMinute);
                hasOverlap = true;
                break; // Stop checking, we found a conflict
            }
        }

        // Check existing FIXED and PLANNED tasks
        if (!hasOverlap) {
            for (const task of timeline.tasks) {
                if (task.type !== 'FIXED' && task.type !== 'PLANNED') continue;

                const taskEnd = task.minuteOffset + task.duration;
                if (overlaps(candidateStart, candidateEnd, task.minuteOffset, taskEnd)) {
                    jumpTo = Math.max(jumpTo, taskEnd);
                    hasOverlap = true;
                    break;
                }
            }
        }

        // Found free slot
        if (!hasOverlap) {
            return candidateStart;
        }

        // Jump past the conflicting item
        candidateStart = jumpTo;
    }

    return null; // No free slot found
}

/**
 * Check if two time intervals overlap.
 *
 * @param {number} start1 - Start of interval 1
 * @param {number} end1 - End of interval 1
 * @param {number} start2 - Start of interval 2
 * @param {number} end2 - End of interval 2
 * @returns {boolean} true if intervals overlap
 */
function overlaps(start1, end1, start2, end2) {
    return start1 < end2 && end1 > start2;
}
```

```javascript
/**
 * Build list of pomodoros from selected work tasks.
 * Converts tasks to 25-minute pomodoros.
 *
 * @param {string[]} selectedPomodoroIds - Task IDs selected by user
 * @returns {Task[]} List of pomodoro tasks (no minuteOffset yet)
 */
function buildPomodoroList(selectedPomodoroIds) {
    const pomodoros = [];

    for (const taskId of selectedPomodoroIds) {
        const task = state.pomodoroTasks.find(t => t.id === taskId);
        if (!task) continue;

        const nbPomodoros = Math.ceil(task.remaining_min / 25);

        for (let i = 0; i < nbPomodoros; i++) {
            pomodoros.push({
                minuteOffset: 0, // Placeholder, will be calculated later
                duration: 25,
                type: 'POMODORO',
                taskId: task.id,
                taskName: task.name,
                isFixed: false,
                pomodoroIndex: i + 1,
                pomodoroTotal: nbPomodoros
            });
        }
    }

    return pomodoros;
}

/**
 * Build list of respirations from user selections (with duplicates).
 *
 * @param {string[]} selectedRespirationIds - Can contain duplicates (x2, x3...)
 * @returns {Task[]} List of respiration tasks (no minuteOffset yet)
 */
function buildRespirationList(selectedRespirationIds) {
    const respirations = [];

    for (const taskId of selectedRespirationIds) {
        const task = state.recurrentTasks.find(t => t.id === taskId);
        if (!task) continue;

        respirations.push({
            minuteOffset: 0, // Placeholder
            duration: task.duration_min,
            type: 'RESPIRATION',
            taskId: task.id,
            taskName: task.name,
            isFixed: false
        });
    }

    return respirations;
}

/**
 * Alternate pomodoros and respirations (cycling respirations if needed).
 *
 * @param {Task[]} pomodoros - List of work tasks
 * @param {Task[]} respirations - List of pause tasks
 * @returns {Task[]} Alternated list [P, R, P, R, P, R...]
 */
function alternatePomodoresRespiration(pomodoros, respirations) {
    const alternated = [];

    // CASE A: No respirations → Mode Focus (only pomodoros)
    if (respirations.length === 0) {
        return [...pomodoros];
    }

    // CASE B: Respirations available → Alternate
    let respirationIndex = 0;

    for (const pomo of pomodoros) {
        alternated.push(pomo);

        // Add respiration (cycling if needed)
        const resp = respirations[respirationIndex % respirations.length];
        alternated.push({...resp}); // Clone to allow multiple instances

        respirationIndex++;
    }

    return alternated;
}

/**
 * Insert calins every 2 respirations (if enabled).
 * Modifies alternatedList in-place.
 *
 * @param {Task[]} alternatedList - List to modify
 * @param {boolean} calinEnabled - User checkbox state
 * @param {number} calinDuration - Duration from recurrent tasks
 */
function insertCalins(alternatedList, calinEnabled, calinDuration = 10) {
    if (!calinEnabled) return;

    let respirationCount = 0;

    for (let i = 0; i < alternatedList.length; i++) {
        if (alternatedList[i].type === 'RESPIRATION') {
            respirationCount++;

            if (respirationCount % 2 === 0) {
                // Insert calin BEFORE this respiration
                alternatedList.splice(i, 0, {
                    minuteOffset: 0,
                    duration: calinDuration,
                    type: 'CALIN',
                    taskId: 'AUTO_CALIN',
                    taskName: 'Câlin',
                    isFixed: false,
                    autoGenerated: true
                });

                i++; // Adjust index after insertion
            }
        }
    }
}

/**
 * Insert clopes based on cumulative duration (if enabled).
 * Modifies alternatedList in-place.
 *
 * @param {Task[]} alternatedList - List to modify
 * @param {boolean} clopeEnabled - User checkbox state
 * @param {number} intervalMin - Cumulative duration threshold (default 120)
 */
function insertClopes(alternatedList, clopeEnabled, intervalMin = 120) {
    if (!clopeEnabled) return;

    let cumulativeDuration = 0;

    for (let i = 0; i < alternatedList.length; i++) {
        cumulativeDuration += alternatedList[i].duration;

        if (cumulativeDuration >= intervalMin) {
            // Insert clope AFTER this task
            alternatedList.splice(i + 1, 0, {
                minuteOffset: 0,
                duration: 5,
                type: 'CLOPE',
                taskId: 'AUTO_CLOPE',
                taskName: 'Pause cigarette',
                isFixed: false,
                autoGenerated: true
            });

            // Reset counter
            cumulativeDuration = 0;

            i++; // Adjust index after insertion
        }
    }
}
```

### PHASE 5: rebuildTimeline Function (TIME CALCULATION Step) (5 heures)

**Objectif**: Fonction centrale qui recalcule tous les minuteOffsets

**Fonction à créer** (Basée sur TIMELINE_LINEAR_SPEC_BACKUP.md):

```javascript
/**
 * Recalculate ALL minuteOffsets for timeline.tasks.
 * Respects obstacles, places tasks sequentially.
 * CENTRAL FUNCTION called after ANY modification to task list.
 *
 * ALGORITHM:
 * 1. Sort tasks by current minuteOffset (preserve order)
 * 2. For each task:
 *    a. Skip obstacles (jump currentMinute past obstacle.endMinute)
 *    b. Assign minuteOffset = currentMinute
 *    c. Advance currentMinute by task.duration
 */
function rebuildTimeline() {
    // STEP 1: Build ORDERED list (preserve current order)
    const sortedTasks = [...state.timeline.tasks].sort((a, b) => a.minuteOffset - b.minuteOffset);

    // STEP 2: Calculate minuteOffset for each task
    let currentMinute = 0; // Start from planningStartTime (minute 0)

    for (const task of sortedTasks) {
        // Skip obstacles (temps_morts)
        let skipObstacle = true;

        while (skipObstacle) {
            skipObstacle = false;

            for (const obstacle of state.timeline.obstacles) {
                const taskEnd = currentMinute + task.duration;

                // Check overlap: currentMinute < obstacle.endMinute AND taskEnd > obstacle.startMinute
                if (currentMinute < obstacle.endMinute && taskEnd > obstacle.startMinute) {
                    console.log(`  Skipping obstacle ${obstacle.startMinute}-${obstacle.endMinute}`);
                    currentMinute = obstacle.endMinute; // Jump past obstacle
                    skipObstacle = true; // Recheck (might overlap another obstacle)
                    break;
                }
            }
        }

        // Assign time
        task.minuteOffset = currentMinute;
        currentMinute += task.duration;
    }
```

```python
@gitfocus_bp.route('/api/v2/gitfocus/pinned-pauses', methods=['GET', 'POST'])
def manage_pinned_pauses():
    """
    GET: Retrieve pinned pause IDs
    POST: Save pinned pause IDs
    """
    config_file = DATA_DIR / 'pinned_pauses.json'

    if request.method == 'GET':
        if config_file.exists():
            with open(config_file, 'r', encoding='utf-8') as f:
                data = json.load(f)
                return jsonify({'success': True, 'pinned_ids': data.get('pinned_ids', [])})
        else:
            return jsonify({'success': True, 'pinned_ids': []})

    elif request.method == 'POST':
        data = request.get_json()
        pinned_ids = data.get('pinned_ids', [])

        # Atomic write
        temp_file = config_file.with_suffix('.tmp')
        with open(temp_file, 'w', encoding='utf-8') as f:
            json.dump({'pinned_ids': pinned_ids}, f, indent=2)
        temp_file.replace(config_file)

        return jsonify({'success': True, 'message': f'Saved {len(pinned_ids)} pinned pauses'})
```

```javascript
/**
 * Convert linear timeline to display format with absolute times.
 *
 * @returns {Array<DisplaySlot>} Sorted display slots with date/time
 */
function convertTimelineToDisplayFormat() {
    const displaySlots = [];

    // Convert tasks
    for (const task of state.timeline.tasks) {
        const start = minutesToDateTime(task.minuteOffset, state.planningStartTime);
        const end = minutesToDateTime(task.minuteOffset + task.duration, state.planningStartTime);

        displaySlots.push({
            date: start.date,
            heure_debut: start.time,
            heure_fin: end.time,
            task_name: task.taskName,
            task_id: task.taskId,
            type: task.type.toLowerCase(),
            duration_min: task.duration,
            pomodoro_index: task.pomodoroIndex,
            pomodoro_total: task.pomodoroTotal,
            is_fixed: task.isFixed || false
        });
    }

    // Convert obstacles (temps_morts) for display
    for (const obstacle of state.timeline.obstacles) {
        const start = minutesToDateTime(obstacle.startMinute, state.planningStartTime);
        const end = minutesToDateTime(obstacle.endMinute, state.planningStartTime);

        displaySlots.push({
            date: start.date,
            heure_debut: start.time,
            heure_fin: end.time,
            task_name: obstacle.originalData.titre,
            task_id: 'TEMPS_MORT',
            type: 'temps_mort',
            duration_min: obstacle.endMinute - obstacle.startMinute,
            is_fixed: true
        });
    }

    // Sort by date + time
    displaySlots.sort((a, b) => {
        if (a.date !== b.date) return a.date.localeCompare(b.date);
        return a.heure_debut.localeCompare(b.heure_debut);
    });

    return displaySlots;
}

/**
 * Refresh planning display in UI (HTML rendering).
 */
function refreshPlanningDisplay() {
    const displaySlots = convertTimelineToDisplayFormat();
    const container = document.getElementById('planning-results');

    if (!container) return;

    // Build HTML
    let html = '<div class="planning-timeline">';

    for (const slot of displaySlots) {
        const typeClass = `slot-${slot.type}`;
        const pomodoroInfo = slot.pomodoro_total ?
            `<span class="pomodoro-index">(${slot.pomodoro_index}/${slot.pomodoro_total})</span>` : '';

        html += `
            <div class="planning-slot ${typeClass}" data-task-id="${slot.task_id}">
                <div class="slot-time">${slot.heure_debut} - ${slot.heure_fin}</div>
                <div class="slot-name">${slot.task_name} ${pomodoroInfo}</div>
                <div class="slot-duration">${slot.duration_min} min</div>
            </div>
        `;
    }

    html += '</div>';

    container.innerHTML = html;

    // Calculate statistics
    const stats = calculateStatsFromTimeline();
    displayStatistics(stats);
}

/**
 * Calculate statistics from timeline.
 */
function calculateStatsFromTimeline() {
    let totalWork = 0;
    let totalPause = 0;
    let pomodoroCount = 0;
    let respirationCount = 0;

    for (const task of state.timeline.tasks) {
        if (task.type === 'POMODORO') {
            totalWork += task.duration;
            pomodoroCount++;
        } else if (['RESPIRATION', 'CALIN', 'CLOPE'].includes(task.type)) {
            totalPause += task.duration;
            if (task.type === 'RESPIRATION') respirationCount++;
        }
    }

    return {
        total_work_min: totalWork,
        total_pause_min: totalPause,
        pomodoro_count: pomodoroCount,
        respiration_count: respirationCount,
        total_tasks: state.timeline.tasks.length
    };
}
```
